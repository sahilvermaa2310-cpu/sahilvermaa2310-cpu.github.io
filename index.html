# HDI vs FLFP — ANOVA
import pandas as pd
from scipy import stats
df = pd.read_csv('flfp_2023.csv')
F, p = stats.f_oneway(*[g['flfp'] for _, g in df.groupby('hdi_group')])
print(f"F={F:.2f}, p={p:.4f}") # F=3.00, p=0.0319
